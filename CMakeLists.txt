cmake_minimum_required(VERSION 3.18)
project(WarpAlign VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
add_library(warpalign INTERFACE)
add_library(WarpAlign::warpalign ALIAS warpalign)
target_compile_features(warpalign INTERFACE cxx_std_17)
target_include_directories(warpalign INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_executable(warpalign-example examples/basic.cpp)
target_link_libraries(warpalign-example PRIVATE warpalign)
if(BUILD_TESTING)
 add_executable(test-core tests/unit/core.cpp)
 target_link_libraries(test-core PRIVATE warpalign)
 add_test(NAME core COMMAND test-core)
endif()
# CPU-only is the portable default. AUTO probes without requiring CUDA headers.
option(WARPALIGN_CUDA "Legacy default for ENABLE_CUDA" OFF)
set(ENABLE_CUDA "${WARPALIGN_CUDA}" CACHE STRING "CUDA build: OFF, ON, or AUTO")
set_property(CACHE ENABLE_CUDA PROPERTY STRINGS OFF ON AUTO)
set(WARPALIGN_CUDA OFF) # effective value; ENABLE_CUDA takes precedence
if(NOT ENABLE_CUDA MATCHES "^(OFF|ON|AUTO)$")
 message(FATAL_ERROR "ENABLE_CUDA must be OFF, ON, or AUTO")
endif()
if(ENABLE_CUDA STREQUAL "ON" OR ENABLE_CUDA STREQUAL "AUTO")
 include(CheckLanguage)
 check_language(CUDA)
 if(CMAKE_CUDA_COMPILER)
  set(WARPALIGN_CUDA ON)
 elseif(ENABLE_CUDA STREQUAL "ON")
  message(FATAL_ERROR "CUDA requested but NVCC was not found; use -DENABLE_CUDA=OFF")
 else()
  message(STATUS "NVCC unavailable; building standalone CPU targets")
 endif()
endif()
if(WARPALIGN_CUDA)
 enable_language(CUDA)
 add_executable(test-cuda tests/cuda/score.cu)
 target_link_libraries(test-cuda PRIVATE warpalign)
 set_target_properties(test-cuda PROPERTIES CUDA_STANDARD 17)
 add_test(NAME cuda-runtime COMMAND test-cuda)
 set_tests_properties(cuda-runtime PROPERTIES SKIP_RETURN_CODE 77)
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS warpalign EXPORT WarpAlignTargets)
install(EXPORT WarpAlignTargets NAMESPACE WarpAlign:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WarpAlign)
include(CMakePackageConfigHelpers)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/WarpAlignConfig.cmake "include(\"\${CMAKE_CURRENT_LIST_DIR}/WarpAlignTargets.cmake\")\n")
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/WarpAlignConfigVersion.cmake COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/WarpAlignConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/WarpAlignConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WarpAlign)

option(WARPALIGN_CUDA_PYTHON "Build optional experimental CUDA Python module" OFF)
if(WARPALIGN_CUDA_PYTHON)
 if(NOT WARPALIGN_CUDA)
  message(FATAL_ERROR "WARPALIGN_CUDA_PYTHON requires WARPALIGN_CUDA=ON")
 endif()
 find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
 execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE pybind11_status)
 if(NOT pybind11_status EQUAL 0)
  message(FATAL_ERROR "Install pybind11 into the selected Python environment")
 endif()
 find_package(pybind11 CONFIG REQUIRED)
 pybind11_add_module(_cuda python/cuda_bindings.cu)
 target_link_libraries(_cuda PRIVATE warpalign)
 set_target_properties(_cuda PROPERTIES CUDA_STANDARD 17 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/python/warpalign")
 install(TARGETS _cuda LIBRARY DESTINATION warpalign)
endif()
